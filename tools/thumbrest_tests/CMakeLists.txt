cmake_minimum_required(VERSION 3.24)
project(ThumbRestInputTests LANGUAGES CXX)
enable_testing()
find_path(THUMBREST_OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
set(frame_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/Runtimes/OpenXRFrameLoop.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${frame_source}")
file(READ "${frame_source}" frame_code)
# Compile the actual action reads/merge, not a duplicate Boolean implementation.
function(extract_input begin_marker end_marker output)
    string(FIND "${frame_code}" "${begin_marker}" begin)
    string(FIND "${frame_code}" "${end_marker}" end)
    if(begin LESS 0 OR end LESS begin)
        message(FATAL_ERROR "Production input markers changed")
    endif()
    math(EXPR length "${end}-${begin}")
    string(SUBSTRING "${frame_code}" ${begin} ${length} code)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${code}")
endfunction()
extract_input("const bool thumbRestClick=" "auto getVec2 =" thumbrest_read.inc)
extract_input("const bool physicalClick=" "auto& uiHand=uiTracking.hands[i];" thumbrest_merge.inc)
add_executable(thumbrest_tests main.cpp)
target_include_directories(thumbrest_tests PRIVATE ../../include "${THUMBREST_OPENXR_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(thumbrest_tests PRIVATE cxx_std_20)
target_compile_definitions(thumbrest_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME thumbrest_input COMMAND thumbrest_tests)
