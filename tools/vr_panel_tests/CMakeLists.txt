cmake_minimum_required(VERSION 3.24)
project(VrPanelTests LANGUAGES CXX)
enable_testing()
find_path(PANEL_OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
set(VR_PANEL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/Overlay/VrPanel.cpp" CACHE FILEPATH "Panel source for regression control")
add_executable(vr_panel_tests main.cpp "${VR_PANEL_SOURCE}" ../../src/Render/ColorBlit.cpp)
target_include_directories(vr_panel_tests PRIVATE ../../include "${PANEL_OPENXR_INCLUDE_DIR}")
target_compile_features(vr_panel_tests PRIVATE cxx_std_20)
target_compile_definitions(vr_panel_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D12)
target_link_libraries(vr_panel_tests PRIVATE d3d12 dxgi d3dcompiler)
add_test(NAME vr_panel_sync_warp COMMAND vr_panel_tests)
set_tests_properties(vr_panel_sync_warp PROPERTIES TIMEOUT 30)
